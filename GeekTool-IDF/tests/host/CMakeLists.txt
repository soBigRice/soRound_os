cmake_minimum_required(VERSION 3.20)
project(geektool_performance_tests LANGUAGES C)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory(../../managed_components/lvgl__lvgl lvgl)
add_executable(performance_tests performance_tests.c countdown_tests.c ../../main/glyph.c)
target_include_directories(performance_tests PRIVATE stubs ../../main)
target_link_libraries(performance_tests PRIVATE lvgl m)
target_compile_options(performance_tests PRIVATE -Wall -Wextra -Wno-missing-field-initializers)
enable_testing()
add_test(NAME performance_regressions COMMAND performance_tests)
add_executable(ota_ui_tests ota_ui_tests.c ../../main/glyph.c ../../main/i18n.c ../../main/font_cn16.c)
target_include_directories(ota_ui_tests PRIVATE ../ota/stubs ../../main)
target_link_libraries(ota_ui_tests PRIVATE lvgl m)
add_test(NAME ota_ui_states COMMAND ota_ui_tests)
add_executable(buttons_tests buttons_tests.c ../../main/buttons.c ../../main/glyph.c)
target_include_directories(buttons_tests PRIVATE stubs ../../main)
target_link_libraries(buttons_tests PRIVATE lvgl m)
target_compile_options(buttons_tests PRIVATE -Wall -Wextra -Wno-missing-field-initializers)
add_test(NAME physical_button_mapping COMMAND buttons_tests)
add_executable(hid_tests ../hid/hid_tests.c ../hid/hid_fake.c)
target_include_directories(hid_tests PRIVATE ../hid/stubs stubs ../../main)
target_compile_options(hid_tests PRIVATE -Wall -Wextra -Werror)
add_test(NAME remote_hid_protocol COMMAND hid_tests)
add_executable(remote_ui_tests ../hid/remote_ui_tests.c ../hid/hid_fake.c ../../main/ble_hid.c
               ../../main/glyph.c ../../main/i18n.c ../../main/font_cn16.c)
target_include_directories(remote_ui_tests PRIVATE ../hid/stubs stubs ../../main)
target_link_libraries(remote_ui_tests PRIVATE lvgl m)
target_compile_options(remote_ui_tests PRIVATE -Wall -Wextra -Werror)
add_test(NAME remote_controls_and_layout COMMAND remote_ui_tests)
