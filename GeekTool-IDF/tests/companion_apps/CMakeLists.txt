cmake_minimum_required(VERSION 3.20)
project(companion_apps_tests LANGUAGES C)
enable_testing()
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../host/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory("../../managed_components/lvgl__lvgl" lvgl)
add_executable(companion_state_tests state_tests.c immediate_persistence.c ../../main/garden_state.c ../../main/focus_state.c)
target_include_directories(companion_state_tests PRIVATE stubs ../../main)
target_compile_options(companion_state_tests PRIVATE -UNDEBUG -Wall -Wextra -Werror)
add_test(NAME garden_focus_state_and_storage COMMAND companion_state_tests)
add_executable(companion_ui_tests ui_tests.c immediate_persistence.c ../../main/garden_state.c ../../main/focus_state.c
    ../../main/i18n.c ../../main/font_cn16.c ../../main/font_companion.c)
target_include_directories(companion_ui_tests PRIVATE stubs ../host/stubs ../../main)
target_compile_options(companion_ui_tests PRIVATE -UNDEBUG -Wall -Wextra -Werror -Wno-missing-field-initializers)
target_link_libraries(companion_ui_tests PRIVATE lvgl m)
add_test(NAME garden_focus_native_ui COMMAND companion_ui_tests)
